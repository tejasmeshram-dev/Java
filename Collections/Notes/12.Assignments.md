import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Deque;
import java.util.HashMap;
import java.util.HashSet;
import java.util.List;
import java.util.PriorityQueue;
import java.util.Queue;

public class Day2Assignment {

    // A. ArrayList operations
    public static List<Integer> arrayListOperations() {

        ArrayList<Integer> list = new ArrayList<>();

        list.add(10);
        list.add(20);
        list.add(30);
        list.add(40);
        list.add(50);

        list.set(1, 99);

        list.remove(list.size() - 1);

        return list;
    }


    // B. Frequency counter
    public static HashMap<Integer, Integer> countFrequencies(
            int[] nums) {

        HashMap<Integer, Integer> freq = new HashMap<>();

        for (int num : nums) {

            freq.put(
                num,
                freq.getOrDefault(num, 0) + 1
            );
        }

        return freq;
    }


    // C. Duplicate detection
    public static boolean containsDuplicate(int[] nums) {

        HashSet<Integer> seen = new HashSet<>();

        for (int num : nums) {

            if (!seen.add(num)) {
                return true;
            }
        }

        return false;
    }


    // D. Valid parentheses
    public static boolean isValid(String s) {

        Deque<Character> stack = new ArrayDeque<>();

        for (int i = 0; i < s.length(); i++) {

            char ch = s.charAt(i);

            if (ch == '(' || ch == '[' || ch == '{') {

                stack.push(ch);

            } else {

                if (stack.isEmpty()) {
                    return false;
                }

                char top = stack.pop();

                if (ch == ')' && top != '(') {
                    return false;
                }

                if (ch == ']' && top != '[') {
                    return false;
                }

                if (ch == '}' && top != '{') {
                    return false;
                }
            }
        }

        return stack.isEmpty();
    }


    // E. Queue implemented using two stacks
    static class MyQueue {

        private final Deque<Integer> input =
                new ArrayDeque<>();

        private final Deque<Integer> output =
                new ArrayDeque<>();

        public void push(int x) {
            input.push(x);
        }

        public int pop() {

            moveIfNeeded();

            return output.pop();
        }

        public int peek() {

            moveIfNeeded();

            return output.peek();
        }

        public boolean empty() {

            return input.isEmpty() && output.isEmpty();
        }

        private void moveIfNeeded() {

            if (output.isEmpty()) {

                while (!input.isEmpty()) {
                    output.push(input.pop());
                }
            }
        }
    }


    // F. Kth largest element
    public static int findKthLargest(int[] nums, int k) {

        if (k < 1 || k > nums.length) {
            throw new IllegalArgumentException(
                "k must be between 1 and nums.length"
            );
        }

        PriorityQueue<Integer> minHeap =
                new PriorityQueue<>();

        for (int num : nums) {

            minHeap.offer(num);

            if (minHeap.size() > k) {
                minHeap.poll();
            }
        }

        return minHeap.peek();
    }


    // G. Linked-list node
    static class ListNode {

        int val;
        ListNode next;

        ListNode(int val) {
            this.val = val;
        }
    }


    // G. Reverse a linked list
    public static ListNode reverseList(ListNode head) {

        ListNode prev = null;
        ListNode current = head;

        while (current != null) {

            ListNode next = current.next;

            current.next = prev;

            prev = current;

            current = next;
        }

        return prev;
    }


    // Utility: Convert an array to a linked list
    public static ListNode buildList(int[] nums) {

        ListNode dummy = new ListNode(0);
        ListNode current = dummy;

        for (int num : nums) {

            current.next = new ListNode(num);

            current = current.next;
        }

        return dummy.next;
    }


    // Utility: Print a linked list
    public static void printList(ListNode head) {

        ListNode current = head;

        while (current != null) {

            System.out.print(current.val + " -> ");

            current = current.next;
        }

        System.out.println("null");
    }


    // Basic testing
    public static void main(String[] args) {

        // A. ArrayList
        System.out.println(
            "ArrayList: " + arrayListOperations()
        );


        // B. Frequency counting
        int[] frequencyInput = {1, 2, 2, 3, 3, 3};

        System.out.println(
            "Frequencies: " +
            countFrequencies(frequencyInput)
        );


        // C. Duplicate detection
        System.out.println(
            "Has duplicate: " +
            containsDuplicate(new int[]{1, 2, 3, 1})
        );

        System.out.println(
            "Has duplicate: " +
            containsDuplicate(new int[]{1, 2, 3, 4})
        );


        // D. Parentheses
        System.out.println(
            "Valid parentheses: " +
            isValid("()[]{}")
        );

        System.out.println(
            "Valid parentheses: " +
            isValid("([)]")
        );


        // E. Queue
        MyQueue queue = new MyQueue();

        queue.push(10);
        queue.push(20);
        queue.push(30);

        System.out.println(
            "Queue peek: " + queue.peek()
        );

        System.out.println(
            "Queue pop: " + queue.pop()
        );

        System.out.println(
            "Queue peek: " + queue.peek()
        );

        System.out.println(
            "Queue empty: " + queue.empty()
        );


        // F. Kth largest
        System.out.println(
            "Kth largest: " +
            findKthLargest(
                new int[]{3, 2, 1, 5, 6, 4},
                2
            )
        );


        // G. Reverse linked list
        ListNode head = buildList(
            new int[]{1, 2, 3, 4, 5}
        );

        System.out.print("Original list: ");
        printList(head);

        head = reverseList(head);

        System.out.print("Reversed list: ");
        printList(head);
    }

One important limitation of the reference implementation

The queue implementation assumes that pop() and peek() are only called when the queue is nonempty. That matches the usual constraints of the corresponding interview exercise.

For a production-quality queue, you would define explicit behavior for empty-queue operations.

Similarly, isValid() assumes its input contains only the three supported bracket types. If arbitrary characters are allowed, you should specify how they should be handled and implement that behavior.

This distinction between a solution designed for a particular coding problem and a robust general-purpose API is useful for Java development interviews.
